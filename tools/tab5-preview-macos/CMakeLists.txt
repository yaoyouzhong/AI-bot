cmake_minimum_required(VERSION 3.20)
project(tab5_native_preview C CXX)
set(CMAKE_C_STANDARD 11)
if(NOT DEFINED TAB5_SOURCE OR NOT EXISTS "${TAB5_SOURCE}/firmware/main/ui.c")
    message(FATAL_ERROR "Set TAB5_SOURCE to the independently verified TAB5 source directory")
endif()
set(ROOT "${TAB5_SOURCE}")
set(PREVIEW "${ROOT}/tests/ui-preview")
set(LVGL "${ROOT}/firmware/managed_components/lvgl__lvgl")
file(GLOB_RECURSE LV_SRC "${LVGL}/src/*.c")
add_library(lvgl STATIC ${LV_SRC})
set_source_files_properties("${LVGL}/src/draw/sw/lv_draw_sw_transform.c" PROPERTIES COMPILE_DEFINITIONS "lv_draw_sw_transform=tab5_lvgl_reference_transform")
target_sources(lvgl PRIVATE "${ROOT}/firmware/main/draw_orthogonal.c")
target_compile_definitions(lvgl PRIVATE TAB5_TRANSFORM_PREVIEW)
target_include_directories(lvgl PUBLIC "${LVGL}" "${PREVIEW}")
target_compile_definitions(lvgl PUBLIC LV_CONF_INCLUDE_SIMPLE _CRT_SECURE_NO_WARNINGS LV_USE_STDLIB_STRING=LV_STDLIB_CLIB LV_DRAW_SW_SUPPORT_L8=0 LV_DRAW_SW_SUPPORT_AL88=0 LV_DRAW_SW_SUPPORT_I1=0)
add_executable(ui-preview "${PREVIEW}/preview.c" "${ROOT}/firmware/main/tab5_font_24.c" "${ROOT}/firmware/main/tab5_font_30.c" "${ROOT}/firmware/main/tab5_digits_72.c" "${ROOT}/firmware/main/tab5_digits_96.c" "${ROOT}/firmware/main/tab5_timer_digits_192.c" "${ROOT}/firmware/main/link_state.c" "${ROOT}/.tools/esp-idf/components/json/cJSON/cJSON.c")
target_sources(ui-preview PRIVATE "${ROOT}/firmware/main/codex_companion.c" "${ROOT}/firmware/main/claude_companion.c" "${CMAKE_CURRENT_LIST_DIR}/native_pinyin.c" "${ROOT}/firmware/main/tab5_ime.c" "${ROOT}/firmware/main/tab5_pinyin_data.c" "${ROOT}/firmware/main/gallery_buffer.c")
target_sources(ui-preview PRIVATE "${ROOT}/firmware/main/tab5_annual_digits_144.c" "${ROOT}/firmware/main/tab5_annual_text_26.c" "${ROOT}/firmware/main/tab5_calendar_text_20.c" "${ROOT}/firmware/main/tab5_calendar_month_48.c" "${ROOT}/firmware/main/tab5_calendar_verse_24.c" "${ROOT}/firmware/main/tab5_calendar_meta_18.c" "${PREVIEW}/calendar_compare_regular_24.c" "${PREVIEW}/calendar_compare_running_26.c")
target_sources(ui-preview PRIVATE "${ROOT}/firmware/main/tab5_calendar_brush_24.c" "${ROOT}/firmware/main/tab5_calendar_running_28.c" "${ROOT}/firmware/main/tab5_countdown_24.c" "${ROOT}/firmware/main/tab5_date_serif_data.c" "${ROOT}/firmware/main/agenda_core.c" "${ROOT}/firmware/main/wifi_profiles.c" "${ROOT}/firmware/main/reply_document.c" "${ROOT}/firmware/main/draw_cache.c")
target_include_directories(ui-preview PRIVATE "${ROOT}/firmware/main" "${ROOT}/.tools/esp-idf/components/json/cJSON" "${ROOT}/tests/pinyin/heap-stubs" "${ROOT}/.tools/esp-idf/components/esp_rom/include" "${ROOT}/.tools/esp-idf/components/esp_rom/esp32p4")
target_compile_definitions(ui-preview PRIVATE TAB5_LOCAL_ART=1 TAB5_CODEX_PET_ASSET=1)
target_compile_options(ui-preview PRIVATE -UNDEBUG)
target_link_libraries(ui-preview PRIVATE lvgl z m)
target_compile_options(ui-preview PRIVATE -include "${CMAKE_CURRENT_LIST_DIR}/native_timing.h")

cmake_minimum_required(VERSION 3.20)
project(pinyin_regression C)
set(CMAKE_C_STANDARD 11)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${ROOT}/firmware/main/tab5_ime.c" "${BASELINE}/firmware/main/tab5_ime.c")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/prepare.py" "${ROOT}" "${BASELINE}" "${CMAKE_CURRENT_BINARY_DIR}" COMMAND_ERROR_IS_FATAL ANY)
add_executable(pinyin-test test.c data-under-test.c "${CMAKE_CURRENT_BINARY_DIR}/baseline.c" "${CMAKE_CURRENT_BINARY_DIR}/current.c")
target_include_directories(pinyin-test PRIVATE "${CMAKE_CURRENT_LIST_DIR}" "${ROOT}/firmware/main" "${CMAKE_CURRENT_BINARY_DIR}" "${ROOT}/.tools/esp-idf/components/esp_rom/include" "${ROOT}/.tools/esp-idf/components/esp_rom/esp32p4")
target_compile_options(pinyin-test PRIVATE /UNDEBUG /utf-8)

cmake_minimum_required(VERSION 3.16)
project(display_bandwidth_policy C)
set(CMAKE_C_STANDARD 11)
if(NOT EXISTS "${IDF_PATH}/components/soc/esp32p4/register/soc/icm_sys_struct.h")
  message(FATAL_ERROR "Pass the pinned ESP-IDF path with -DIDF_PATH=...")
endif()
foreach(case fixed baseline)
  if(case STREQUAL "fixed")
    set(source "${CMAKE_CURRENT_LIST_DIR}/../../firmware/main")
  else()
    set(source "${BASELINE_SOURCE}/firmware/main")
  endif()
  add_executable(policy-${case} policy_test.c "${source}/display_bandwidth.c")
  target_include_directories(policy-${case} PRIVATE stubs "${source}"
    "${IDF_PATH}/components/soc/esp32p4/register")
  if(case STREQUAL "baseline")
    target_compile_definitions(policy-${case} PRIVATE TEST_BASELINE=1)
  endif()
endforeach()
